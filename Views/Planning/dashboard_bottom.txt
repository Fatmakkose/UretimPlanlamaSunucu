        var ordersData = @Html.Raw(Json.Serialize(Model));

        function formatDateForInput(dateStr) {
            if (!dateStr) return '';
            var d = new Date(dateStr);
            if (isNaN(d.getTime())) return '';
            var month = '' + (d.getMonth() + 1),
                day = '' + d.getDate(),
                year = d.getFullYear();

            if (month.length < 2) month = '0' + month;
            if (day.length < 2) day = '0' + day;

            return [year, month, day].join('-');
        }

        function filterOrders() {
            var q = $('#orderSearch').val().toLowerCase();
            $('.order-card').each(function() {
                var text = $(this).text().toLowerCase();
                $(this).toggle(text.indexOf(q) > -1);
            });
        }

        function switchTab(tab) {
            currentTab = tab;
            
            var path = '/Planning/';
            if(tab === 'sample') path += 'SampleTest';
            if(tab === 'purchasing') path += 'Purchasing';
            if(tab === 'production') path += 'Production';
            
            var url = new URL(window.location.origin + path);
            var searchParams = new URLSearchParams(window.location.search);
            if(searchParams.has('orderId')) {
                url.searchParams.set('orderId', searchParams.get('orderId'));
            }
            window.history.pushState({}, '', url);

            $('.phase-card').removeClass('active');
            $('#nav-' + tab).addClass('active');

            $('.tab-content').removeClass('active');
            $('#tab-' + tab).addClass('active');
        }

        function savePlanningData(e) {
            e.preventDefault();
            var formData = new FormData(document.getElementById('orderPlanningForm'));
            var obj = {};
            formData.forEach((value, key) => {
                if (value !== '') {
                    obj[key] = value;
                }
            });

            $.ajax({
                url: '/Planning/UpdateOrderPlanning',
                type: 'POST',
                contentType: 'application/json',
                data: JSON.stringify(obj),
                success: function (res) {
                    if (res.success) {
                        alert("Planlama başarıyla kaydedildi.");
                        // Refresh the local array data
                        var index = ordersData.findIndex(x => x.id == obj.Id);
                        if(index > -1) {
                            Object.assign(ordersData[index], obj);
                        }
                    } else {
                        alert("Hata: " + res.message);
                    }
                },
                error: function (err) {
                    console.error(err);
                    alert("Kayıt sırasında bir hata oluştu.");
                }
            });
        }
